cmake_minimum_required(VERSION 3.20)
project(EditorFrameworkConsumer LANGUAGES CXX)
find_package(EditorFramework CONFIG REQUIRED)
find_package(GameFeature CONFIG REQUIRED)
option(EDITOR_FRAMEWORK_CONSUMER_WITH_IMGUI "Also consume the optional EditorImGui package" ON)
if(EDITOR_FRAMEWORK_CONSUMER_WITH_IMGUI)
    find_package(EditorImGui CONFIG REQUIRED)
endif()
add_executable(EditorFrameworkConsumer main.cpp)
target_compile_features(EditorFrameworkConsumer PRIVATE cxx_std_23)
target_link_libraries(EditorFrameworkConsumer PRIVATE Aether::EditorFramework Aether::GameFeature)
if(EDITOR_FRAMEWORK_CONSUMER_WITH_IMGUI)
    target_link_libraries(EditorFrameworkConsumer PRIVATE Aether::EditorImGui)
endif()
